## Levantamento de requisitos

O sistema proposto organiza alunos, instrutores, atividades, turmas, matrículas e ausências de uma academia de ginástica. O levantamento foi feito pela análise da proposta da atividade, onde:

- Os alunos podem participar de várias turmas;
- Cada turma possui uma atividade, um instrutor e um aluno monitor;
- Um instrutor pode orientar várias turmas, enquanto um aluno monitora no máximo uma turma.


### Modelagem

- O monitor deve estar matriculado na turma que acompanha;
- O número de alunos é calculado pelas matrículas, sem contador editável;
- Ausências são registradas por aluno, turma e data; considera-se até uma aula por turma em cada data;
- Altura é informada em metros, peso (em quilogramas) e duração em minutos;
- O cadastro do aluno na academia é diferente de sua inscrição em uma turma.

## Regras de negócio

| ID | Regra |
|---|---|
| RN01 | Toda turma possui exatamente um instrutor |
| RN02 | Um instrutor pode orientar várias turmas, inclusive de atividades diferentes |
| RN03 | Cada turma corresponde a um tipo de atividade |
| RN04 | Cada turma possui exatamente um aluno monitor |
| RN05 | Um aluno pode monitorar no máximo uma turma |
| RN06 | Um aluno pode participar de várias turmas, e uma turma pode receber vários alunos |
| RN07 | As ausências pertencem à matrícula do aluno na turma |
| RN08 | O mesmo aluno não pode possuir duas matrículas na mesma turma |
| RN09 | O monitor deve possuir matrícula na própria turma |
| RN10 | O número de alunos da turma corresponde à quantidade de matrículas existentes |

OBS.:
- RN01–RN07 vêm do cenário; 
- RN08–RN10 explicitam as decisões de modelagem do projeto;
- A identificação da ausência por data também é uma decisão de modelagem.

##  DDL - SQL

```sql

BEGIN;

CREATE SCHEMA academia;
SET search_path TO academia, public;

CREATE TABLE aluno (
    aluno_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    codigo_matricula VARCHAR(20) NOT NULL UNIQUE,
    data_matricula DATE NOT NULL,
    nome VARCHAR(120) NOT NULL,
    endereco TEXT NOT NULL,
    telefone VARCHAR(25) NOT NULL,
    data_nascimento DATE NOT NULL,
    altura_m NUMERIC(4,2) NOT NULL,
    peso_kg NUMERIC(6,2) NOT NULL,
    CONSTRAINT ck_aluno_textos CHECK (
        btrim(codigo_matricula) <> '' AND btrim(nome) <> ''
        AND btrim(endereco) <> '' AND btrim(telefone) <> ''),
    CONSTRAINT ck_aluno_altura CHECK (altura_m > 0),
    CONSTRAINT ck_aluno_peso CHECK (peso_kg > 0),
    CONSTRAINT ck_aluno_datas CHECK (
        data_nascimento <= data_matricula)
);

CREATE TABLE instrutor (
    instrutor_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    rg VARCHAR(30) NOT NULL UNIQUE,
    nome VARCHAR(120) NOT NULL,
    data_nascimento DATE NOT NULL,
    titulacao VARCHAR(120) NOT NULL,
    CONSTRAINT ck_instrutor_textos CHECK (
        btrim(rg) <> '' AND btrim(nome) <> ''
        AND btrim(titulacao) <> '')
);

CREATE TABLE instrutor_telefone (
    instrutor_id BIGINT NOT NULL,
    telefone VARCHAR(25) NOT NULL,
    CONSTRAINT pk_instrutor_telefone
        PRIMARY KEY (instrutor_id, telefone),
    CONSTRAINT fk_telefone_instrutor
        FOREIGN KEY (instrutor_id)
        REFERENCES instrutor (instrutor_id) ON DELETE CASCADE,
    CONSTRAINT ck_telefone_preenchido
        CHECK (btrim(telefone) <> '')
);

CREATE TABLE atividade (
    atividade_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(80) NOT NULL UNIQUE,
    CONSTRAINT ck_atividade_nome CHECK (btrim(nome) <> '')
);

CREATE TABLE turma (
    turma_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(80) NOT NULL,
    horario_aula TIME NOT NULL,
    duracao_minutos SMALLINT NOT NULL,
    data_inicio DATE NOT NULL,
    data_fim DATE NOT NULL,
    atividade_id BIGINT NOT NULL,
    instrutor_id BIGINT NOT NULL,
    monitor_aluno_id BIGINT NOT NULL,
    CONSTRAINT uq_monitor_uma_turma UNIQUE (monitor_aluno_id),
    CONSTRAINT ck_turma_nome CHECK (btrim(nome) <> ''),
    CONSTRAINT ck_turma_duracao CHECK (duracao_minutos > 0),
    CONSTRAINT ck_turma_periodo CHECK (data_fim >= data_inicio),
    CONSTRAINT fk_turma_atividade
        FOREIGN KEY (atividade_id)
        REFERENCES atividade (atividade_id) ON DELETE RESTRICT,
    CONSTRAINT fk_turma_instrutor
        FOREIGN KEY (instrutor_id)
        REFERENCES instrutor (instrutor_id) ON DELETE RESTRICT
);

CREATE TABLE matricula_turma (
    turma_id BIGINT NOT NULL,
    aluno_id BIGINT NOT NULL,
    data_inscricao DATE NOT NULL,
    CONSTRAINT pk_matricula_turma PRIMARY KEY (turma_id, aluno_id),
    CONSTRAINT fk_matricula_turma
        FOREIGN KEY (turma_id)
        REFERENCES turma (turma_id)
        DEFERRABLE INITIALLY DEFERRED,
    CONSTRAINT fk_matricula_aluno
        FOREIGN KEY (aluno_id)
        REFERENCES aluno (aluno_id) ON DELETE RESTRICT
);

ALTER TABLE turma ADD CONSTRAINT fk_monitor_matriculado
    FOREIGN KEY (turma_id, monitor_aluno_id)
    REFERENCES matricula_turma (turma_id, aluno_id)
    DEFERRABLE INITIALLY DEFERRED;

CREATE TABLE ausencia (
    turma_id BIGINT NOT NULL,
    aluno_id BIGINT NOT NULL,
    data_aula DATE NOT NULL,
    CONSTRAINT pk_ausencia
        PRIMARY KEY (turma_id, aluno_id, data_aula),
    CONSTRAINT fk_ausencia_matricula
        FOREIGN KEY (turma_id, aluno_id)
        REFERENCES matricula_turma (turma_id, aluno_id)
        ON DELETE RESTRICT
);

CREATE VIEW vw_turma_resumo AS
SELECT
    t.turma_id, t.nome, t.horario_aula, t.duracao_minutos,
    t.data_inicio, t.data_fim,
    t.atividade_id, a.nome AS atividade,
    t.instrutor_id, i.nome AS instrutor,
    t.monitor_aluno_id, mo.nome AS monitor,
    (
        SELECT COUNT(*)
        FROM matricula_turma m
        WHERE m.turma_id = t.turma_id
    ) AS numero_alunos
FROM turma t
JOIN atividade a ON a.atividade_id = t.atividade_id
JOIN instrutor i ON i.instrutor_id = t.instrutor_id
JOIN aluno mo ON mo.aluno_id = t.monitor_aluno_id;

COMMIT;
```
